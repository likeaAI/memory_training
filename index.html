💻 Unsloth + TRL DPOTrainer 강화학습 실행 코드

단순 SFT(무조건 암기)가 아니라, 실제 내 뇌에서 반응속도(TTFK)가 빨랐던 연상(`chosen`)과 느렸거나 실패한 연상(`rejected`)의 격차를 학습시키는 진짜 DPO(Direct Preference Optimization) 코드입니다. 

from unsloth import FastLanguageModel, PatchDPOTrainer
PatchDPOTrainer()
from datasets import load_dataset
from trl import DPOTrainer, DPOConfig

# 1. 로컬 모델 로드 (Gemma 4 / Qwen 2.5 4bit)
model, tokenizer = FastLanguageModel.from_pretrained(
    model_name="unsloth/gemma-2-9b-it-bnb-4bit",
    max_seq_length=1024,
    load_in_4bit=True,
)
model = FastLanguageModel.get_peft_model(model, r=16, lora_alpha=16, target_modules=["q_proj", "k_proj", "v_proj", "o_proj"])

# 2. BrainLock NeuroRL에서 내보낸 DPO 선호도 데이터셋 로드
dataset = load_dataset("json", data_files="brainlock_dpo_pairs.jsonl", split="train")

# 3. DPO 강화학습 실행 (반응속도 고보상 Chosen 선호 정렬)
dpo_trainer = DPOTrainer(
    model=model,
    ref_model=None,
    tokenizer=tokenizer,
    train_dataset=dataset,
    beta=0.1,
    args=DPOConfig(per_device_train_batch_size=2, gradient_accumulation_steps=4, num_train_epochs=3, learning_rate=5e-6, output_dir="neuro_rl_dpo_outputs"),
)
dpo_trainer.train()
model.save_pretrained_gguf("brainlock_neurorl_model", tokenizer, quantization_method="q4_k_m")
